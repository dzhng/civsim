import * as THREE from "three/webgpu";
import {vec4} from "three/tsl";
import {createSourceCoverSampler} from "../packages/photoreal-renderer/src/landscape/terrainMaterial";
export async function probe(canvas: HTMLCanvasElement) {
 const renderer=new THREE.WebGPURenderer({canvas,antialias:false});
 await renderer.init();renderer.setSize(99,66);renderer.toneMapping=THREE.NoToneMapping;renderer.outputColorSpace=THREE.LinearSRGBColorSpace;
 const scene=new THREE.Scene(); const camera=new THREE.OrthographicCamera(-1,1,1,-1,0.1,10);camera.position.z=5;
 const values=[255,128,0,0,230,255];
 const {map,cover}=createSourceCoverSampler({data:new Uint8Array(values),width:3,height:2,rect:{min:[-1,-1],max:[1,1]}});
 const geometry=new THREE.PlaneGeometry(2,2);const material=new THREE.MeshBasicNodeMaterial();material.fragmentNode=vec4(cover,cover,cover,1);scene.add(new THREE.Mesh(geometry,material));
 const target=new THREE.RenderTarget(99,66,{depthBuffer:false});renderer.setRenderTarget(target);renderer.render(scene,camera);
 const data=await renderer.readRenderTargetPixelsAsync(target,0,0,99,66);
 const smooth=(x:number)=>{const t=Math.min(1,Math.max(0,(x-.74)/.26));return t*t*(3-2*t)};
 const rowStride=Math.ceil(99*4/256)*256; // Pinned Three returns padded WebGPU rows.
 const pixels=[];
 for(let row=0;row<2;row++)for(let col=0;col<3;col++){const x=16+col*33,y=16+row*33;pixels.push({row,col,source:values[row*3+col],expected:Math.round(smooth(values[row*3+col]/255)*255),actual:Number(data[y*rowStride+x*4])});}
 const interpolation=[];
 for(const [x,y] of [[32,16],[16,32]]) {
  const u=(x+.5)/99*3-.5,v=(y+.5)/66*2-.5;
  const ix=Math.floor(u),iy=Math.floor(v),fx=u-ix,fy=v-iy;
  const at=(cx:number,cy:number)=>values[Math.max(0,Math.min(1,cy))*3+Math.max(0,Math.min(2,cx))];
  const filtered=(at(ix,iy)*(1-fx)+at(ix+1,iy)*fx)*(1-fy)+(at(ix,iy+1)*(1-fx)+at(ix+1,iy+1)*fx)*fy;
  interpolation.push({x,y,filteredBand:filtered,expected:Math.round(smooth(filtered/255)*255),actual:Number(data[y*rowStride+x*4])});
 }
 const texturesBefore=renderer.info.memory.textures;map.dispose();const texturesAfter=renderer.info.memory.textures;
 geometry.dispose();material.dispose();target.dispose();renderer.dispose();
 return {rowStride,readbackBytes:data.byteLength,interpolation,pixels,texturesBefore,texturesAfter,allWithinTwo:[...pixels,...interpolation].every(p=>Math.abs(p.actual-p.expected)<=2),textureReleased:texturesAfter===texturesBefore-1};
}

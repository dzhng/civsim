import { execFileSync } from 'node:child_process';
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { resolve } from 'node:path';
import { APPEARANCE_DESCRIPTORS } from '/Users/david/dev/game-battle-model-quality/packages/soldier-assets/src/appearance.ts';
import { rosterRecipe } from '/Users/david/dev/game-battle-model-quality/packages/soldier-assets/bake/roster.mjs';

const root = '/Users/david/dev/game-battle-model-quality';
const output = '/Users/david/dev/game-catalog-admission/throwaway/catalog-admission/source-reproduction.cdjpuY';
const revision = '2e200452f1fe2ab815923ac1f26000c729a1a636';
const sources = new Map();
const appearances = [];
for (const [id, descriptor] of APPEARANCE_DESCRIPTORS.entries()) {
  const recipe = await rosterRecipe(descriptor);
  const lods = recipe.lods ?? 'lods';
  const name = recipe.source === 'mounted-family' ? lods.split('/')[0] : recipe.source.split('/').at(-1);
  const source = `packages/soldier-assets/assets/source/${recipe.source}/${name}.blend`;
  const body = recipe.source === 'heavy-kit' ? 'HeavyKit-Deform'
    : recipe.source.includes('phalanx') ? 'MediumPhalanx-Deform' : `${name}-Deform`;
  const key = recipe.source === 'mounted-family' ? name : recipe.source.replaceAll('/', '--');
  if (!sources.has(source)) sources.set(source, {
    key, source, body,
    trackedLods: `packages/soldier-assets/assets/source/${recipe.source}/${lods}`,
    targets: [8000, 1000, 800],
    sourceSha256: createHash('sha256').update(await readFile(resolve(root, source))).digest('hex'),
  });
  appearances.push({ id, name: descriptor.name, key, mounted: descriptor.look.mounted, recipe });
}
await mkdir(resolve(output, 'checkout'), { recursive: true });
await mkdir(resolve(output, 'logs'), { recursive: true });
await mkdir(resolve(output, 'products'), { recursive: true });
const archive = resolve(output, 'tracked-inputs.tar');
execFileSync('git', ['archive', '--format=tar', '-o', archive, revision,
  'packages/soldier-assets/bake', 'packages/soldier-assets/src', ...sources.keys()], { cwd: root });
execFileSync('tar', ['-xf', archive, '-C', resolve(output, 'checkout')]);
for (const row of sources.values()) {
  const archived = createHash('sha256').update(await readFile(resolve(output, 'checkout', row.source))).digest('hex');
  if (archived !== row.sourceSha256) throw new Error(`${row.source}: working file differs from pinned tracked source`);
}
const manifest = { revision, root, output, sources: [...sources.values()], appearances };
await writeFile(resolve(output, 'manifest.json'), JSON.stringify(manifest, null, 2) + '\n');
console.log(JSON.stringify({ revision, sources: sources.size, appearances: appearances.length, archive, invariant: 'Fresh extraction of tracked Blender sources and existing exporter/baker code; every source hash equals pinned revision.' }));

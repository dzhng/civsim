import { spawn } from 'node:child_process';
import { createWriteStream } from 'node:fs';
import { readFile, writeFile } from 'node:fs/promises';
import { resolve } from 'node:path';

const output = '/Users/david/dev/game-catalog-admission/throwaway/catalog-admission/source-reproduction.cdjpuY';
const manifest = JSON.parse(await readFile(resolve(output, 'manifest.json')));
const commands = [], results = [];
const pending = manifest.sources.filter(source => source.key !== 'heavy-kit');
const blender = '/Applications/Blender.app/Contents/MacOS/Blender';
async function run(file, args, logfile, key, stage) {
  const command = { key, stage, executable: file, args, cwd: output,
    env: { PYTHONDONTWRITEBYTECODE: '1' }, startedAt: new Date().toISOString(), logfile };
  commands.push(command);
  await writeFile(resolve(output, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
  const log = createWriteStream(logfile);
  const child = spawn(file, args, { cwd: output, env: { ...process.env, PYTHONDONTWRITEBYTECODE: '1' },
    stdio: ['ignore', 'pipe', 'pipe'], timeout: 15 * 60_000 });
  child.stdout.pipe(log); child.stderr.pipe(log);
  const outcome = await new Promise(resolveResult => {
    child.on('error', error => resolveResult({ exitCode: null, error: String(error) }));
    child.on('close', (exitCode, signal) => resolveResult({ exitCode, signal }));
  });
  log.end();
  Object.assign(command, outcome, { finishedAt: new Date().toISOString() });
  await writeFile(resolve(output, 'commands.json'), JSON.stringify(commands, null, 2) + '\n');
  return outcome;
}
async function worker() {
  while (pending.length) {
    const source = pending.shift();
    console.log('START', source.key, new Date().toISOString());
    const exported = await run(blender, [
      '--background', '--factory-startup', '--threads', '2', '--python-exit-code', '1',
      '--python', resolve(output, 'checkout/packages/soldier-assets/bake/blender-mesh-lods.py'), '--',
      '--source', resolve(output, 'checkout', source.source), '--body', source.body,
      '--output', resolve(output, 'products', source.key),
      '--near-triangles', '8000', '--mid-triangles', '1000', '--far-triangles', '800',
    ], resolve(output, 'logs', `${source.key}-export.txt`), source.key, 'export');
    let verified;
    if (exported.exitCode === 0) verified = await run(process.execPath,
      [resolve(output, 'verify-products.mjs'), source.key],
      resolve(output, 'logs', `${source.key}-verify.txt`), source.key, 'bake-load-compare');
    results.push({ key: source.key, exported, verified });
    await writeFile(resolve(output, 'run-results.json'), JSON.stringify(results, null, 2) + '\n');
    console.log('FINISH', source.key, JSON.stringify(results.at(-1)), new Date().toISOString());
  }
}
await Promise.all([worker(), worker()]);
process.exitCode = results.every(row => row.exported.exitCode === 0 && row.verified?.exitCode === 0) ? 0 : 1;
console.log('COMPLETE', results.length, 'remaining sources, exit', process.exitCode);

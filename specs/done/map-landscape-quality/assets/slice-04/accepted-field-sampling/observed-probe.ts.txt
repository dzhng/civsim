// Verification only: accepted CPU producer, unchanged source, bounded Alps window.
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { field } from './harness';
import { buildCampaignLandscape, campaignLandscapeAllocation } from '../packages/game-renderer/src/terrain/campaignLandscape';
import { buildCampaignCoast, campaignCoastCell } from '../packages/game-renderer/src/terrain/campaignCoast';
import { campaignRelief } from '../packages/game-renderer/src/terrain/campaignRelief';
const center: [number, number] = [-450, 990], radius = 64;
const hash = (v: ArrayBufferView) => createHash('sha256').update(new Uint8Array(v.buffer, v.byteOffset, v.byteLength)).digest('hex');
const sourceHashes = () => ({ height: hash(field.height), biome: hash(field.biome), mask: hash(field.renderMask.classes) });
const before = sourceHashes();
const {heightAt} = campaignRelief(field, 2); // fixed production filter, independent of geometry spacing
const cases = [2, 1].map(cell => {
  const a = campaignLandscapeAllocation(radius, cell, campaignCoastCell(field.renderMask));
  const ox = Math.floor((center[0] - radius) / cell) * cell;
  const oy = Math.floor((center[1] - radius) / cell) * cell;
  const coast = buildCampaignCoast((x,y) => !field.renderWaterAt(x,y), ox-a.halo*cell, oy-a.halo*cell, a.paddedSize, cell, campaignCoastCell(field.renderMask));
  const start = performance.now();
  const built = buildCampaignLandscape(field, center, radius, cell);
  const buildMs = performance.now()-start;
  const m = built.surface.mesh;
  let vertexMaxError = 0, checkedVertices = 0, nonInteriorVertices = 0;
  for(let k=0;k<m.vertices.length;k+=10) {
    const x=m.vertices[k], y=m.vertices[k+1];
    if(m.waterCoverage![k/10] || coast.inlandAt(x,y)<16) { nonInteriorVertices++; continue; }
    checkedVertices++;
    vertexMaxError = Math.max(vertexMaxError, Math.abs(m.vertices[k+2]-heightAt(x,y,coast.inlandAt(x,y))));
  }
  return {cell, coast, built, stats:{cell, triangles:m.triangles, vertices:m.vertices.length/10,
    geometryBytes:m.vertices.byteLength+m.indices.byteLength,
    finalMeshTypedBytes:Object.values(m).reduce((sum,v)=>sum+(ArrayBuffer.isView(v)?v.byteLength:0),0),
    generationBytes:built.generationBytes, buildMs, checkedVertices, nonInteriorVertices, vertexMaxError}};
});
function summary(values:number[]) {
  const s=[...values].sort((a,b)=>a-b);
  const q=(p:number)=>s[Math.min(s.length-1,Math.floor(p*s.length))];
  return {count:s.length, mean:s.reduce((a,b)=>a+b,0)/s.length, rms:Math.sqrt(s.reduce((a,b)=>a+b*b,0)/s.length), p50:q(.5), p90:q(.9), p95:q(.95), p99:q(.99), max:q(1)};
}
const rows: {x:number,y:number,z:number, errors:number[]}[]=[];
let excluded=0, maxAnalyticDifference=0, maxInlandDifference=0, minInland=Infinity;
// Same quarter-km interior points, offset off either integer mesh and diagonal.
for(let j=0;j<512;j++) for(let i=0;i<512;i++) {
  const x=center[0]-radius+(i+.37)*.25, y=center[1]-radius+(j+.61)*.25;
  const hits=cases.map(c=>c.built.surface.sampleRendered(x,y));
  if(hits.some(h=>!h)) throw Error('Missing rendered sample');
  const safe=cases.every((c,n)=>{
    const m=c.built.surface.mesh, h=hits[n]!;
    return c.coast.inlandAt(x,y)>=16 && [0,1,2].every(corner=>{
      const k=m.indices[h.triangle*3+corner];
      return !m.waterCoverage![k] && c.coast.inlandAt(m.vertices[k*10],m.vertices[k*10+1])>=16;
    });
  });
  if(!safe) {excluded++; continue;}
  const inland=cases.map(c=>c.coast.inlandAt(x,y));
  minInland=Math.min(minInland,...inland);
  maxInlandDifference=Math.max(maxInlandDifference,Math.abs(inland[0]-inland[1]));
  const analytic=inland.map(d=>heightAt(x,y,d));
  maxAnalyticDifference=Math.max(maxAnalyticDifference,Math.abs(analytic[0]-analytic[1]));
  rows.push({x,y,z:analytic[0],errors:hits.map((h,n)=>Math.abs(h!.position[2]-analytic[n]))});
}
if(maxAnalyticDifference!==0) throw Error('Analytic functions differ at compared samples');
if(cases.some(c=>c.stats.vertexMaxError>1e-5)) throw Error('Interior vertices do not match exact producer');
const heights=rows.map(r=>r.z).sort((a,b)=>a-b), highThreshold=heights[Math.floor(.75*heights.length)];
const result={center,radius,extent:{min:[center[0]-radius,center[1]-radius],max:[center[0]+radius,center[1]+radius]},
  units:'world kilometers; height values are exaggerated presentation kilometers',
  sourceBefore:before,sourceAfter:sourceHashes(),sourceUnchanged:JSON.stringify(before)===JSON.stringify(sourceHashes()),
  sourceGrid:{w:field.w,h:field.h,cell:field.cell},coastCell:campaignCoastCell(field.renderMask),
  sampleCount:rows.length,excluded,minInland,maxInlandDifference,maxAnalyticDifference,highThreshold,
  cases:cases.map((c,n)=>({...c.stats,errors:summary(rows.map(r=>r.errors[n])),upperHeightQuartileErrors:summary(rows.filter(r=>r.z>=highThreshold).map(r=>r.errors[n])),
    worstSamples:[...rows].sort((a,b)=>b.errors[n]-a.errors[n]).slice(0,5)})),
  paths:['web/src/campaign/terrain.ts','packages/game-renderer/src/terrain/campaignLandscape.ts','packages/game-renderer/src/terrain/campaignRelief.ts','packages/game-renderer/src/terrain/shorelineMesh.ts','packages/photoreal-renderer/src/landscape/terrainMaterial.ts'].map(path=>({path,sha256:createHash('sha256').update(readFileSync(path)).digest('hex')}))};
writeFileSync('throwaway/mountain-sampling-cpu-results.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result,null,2));

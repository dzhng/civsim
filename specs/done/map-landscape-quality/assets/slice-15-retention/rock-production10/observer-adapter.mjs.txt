// Scratch extension of the retained production proof. No new browser runner.
import { readFile, writeFile } from 'node:fs/promises';
import { fileURLToPath, pathToFileURL } from 'node:url';

const root = new URL('../', import.meta.url);
const retained = new URL('specs/map-landscape-quality/assets/slice-15-retention/screen-ui-production10/observed-scene.mjs.txt', root);

function installBitmapObserver() {
  const records = [];
  const byBitmap = new WeakMap();
  const blobs = new WeakSet();
  const originalFetch = window.fetch;
  const originalDecode = window.createImageBitmap;
  const originalClose = ImageBitmap.prototype.close;
  let phase = 'campaign';
  let texturePrototype;
  let originalListen;
  const byTexture = new WeakMap();
  // Production emits rock-detail-height-<hash>.png; dev emits the plain name.
  const rockUrl = (url) => /\/rock-detail-height(?:-[^/]+)?\.png(?:\?|$)/.test(url);
  window.fetch = async function (...args) {
    const response = await originalFetch.apply(this, args);
    if (rockUrl(response.url)) {
      const originalBlob = response.blob;
      response.blob = async function (...blobArgs) {
        const blob = await originalBlob.apply(this, blobArgs);
        blobs.add(blob);
        return blob;
      };
    }
    return response;
  };
  window.createImageBitmap = async function (...args) {
    const bitmap = await originalDecode.apply(this, args);
    if (blobs.has(args[0])) {
      const record = {
        id: records.length + 1, phase, width: bitmap.width, height: bitmap.height,
        closes: 0, disposals: 0, bitmap: new WeakRef(bitmap), texture: null,
        closedDimensions: null,
      };
      records.push(record);
      byBitmap.set(bitmap, record);
    }
    return bitmap;
  };
  ImageBitmap.prototype.close = function (...args) {
    const record = byBitmap.get(this);
    const result = originalClose.apply(this, args);
    if (record) {
      record.closes++;
      record.closedDimensions = [this.width, this.height];
    }
    return result;
  };
  function observeTexture(texture) {
    const record = byBitmap.get(texture.image);
    if (!record || byTexture.has(texture)) return;
    if (record.texture) throw new Error('Rock bitmap has more than one texture owner');
    record.texture = new WeakRef(texture);
    byTexture.set(texture, record);
    // No strong reference to the texture/bitmap/world in this callback.
    originalListen.call(texture, 'dispose', () => { record.disposals++; });
  }
  window.__rockRetirement = {
    phase: (value) => { phase = value; },
    attach: (campaignWorld) => {
      const texture = campaignWorld.rockDetailMap;
      if (!texture?.image || !byBitmap.has(texture.image)) {
        throw new Error('Campaign rock bitmap was not observed; check production asset URL');
      }
      if (!texturePrototype) {
        texturePrototype = Object.getPrototypeOf(texture);
        originalListen = texturePrototype.addEventListener;
        texturePrototype.addEventListener = function (type, listener) {
          if (type === 'dispose') observeTexture(this);
          return originalListen.call(this, type, listener);
        };
      }
      observeTexture(texture);
      return byTexture.get(texture).id;
    },
    sample: () => records.map(({ bitmap, texture, ...record }) => ({
      ...record, textureObserved: Boolean(texture),
      bitmapAlive: Boolean(bitmap.deref()), textureAlive: Boolean(texture?.deref()),
    })),
    restore: () => {
      window.fetch = originalFetch;
      window.createImageBitmap = originalDecode;
      ImageBitmap.prototype.close = originalClose;
      if (texturePrototype) texturePrototype.addEventListener = originalListen;
    },
  };
}

// Uses the real campaign factory through its live instance, no module import or
// product hook. The getter rejects after the map decode, in the constructor.
async function setupFailure(page, ctx) {
  const result = await page.evaluate(async () => {
    const factory = window.__campaign.rendererOwner().world.constructor;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = 8;
    const sentinel = new Error('rock-lifetime intentional campaign setup failure');
    window.__rockRetirement.phase('campaign-setup-failure');
    let rejected = false;
    try {
      const unexpected = await factory.create(canvas, {
        get fogAt() { throw sentinel; },
      });
      unexpected.dispose();
    } catch (error) {
      rejected = error === sentinel;
    } finally {
      window.__rockRetirement.phase('campaign');
      canvas.remove();
    }
    return { rejected };
  });
  await page.requestGC();
  const records = await page.evaluate(() => window.__rockRetirement.sample().filter(item => item.phase === 'campaign-setup-failure'));
  ctx.check('campaign constructor failure disposes rock texture and closes bitmap exactly once',
    result.rejected && records.length === 1 && records.every(item =>
      item.textureObserved && item.disposals === 1 && item.closes === 1 &&
      item.closedDimensions?.every(size => size === 0)), JSON.stringify({ ...result, records }));
  ctx.check('campaign constructor failure rock texture and bitmap collect',
    records.length === 1 && records.every(item => !item.textureAlive && !item.bitmapAlive), JSON.stringify(records));
}

function replaceOnce(source, before, after) {
  if (source.split(before).length !== 2) throw new Error(`Retained proof anchor is not unique: ${before}`);
  return source.replace(before, after);
}

export async function materialize() {
  let source = await readFile(retained, 'utf8');
  source = replaceOnce(source,
    '"/Users/david/dev/game-map-landscape-quality/web/scenes/worlds.mjs"',
    JSON.stringify(new URL('web/scenes/worlds.mjs', root).href));
  source = replaceOnce(source, 'name: "renderer-lifecycle"', 'name: "rock-lifetime-proof"');
  source = replaceOnce(source, 'export async function run(ctx) {', `export async function run(ctx) {
  const originalContext = ctx;
  ctx = Object.assign(Object.create(ctx), {
    newPage: async (...args) => {
      const page = await originalContext.newPage(...args);
      await page.addInitScript(${installBitmapObserver.toString()});
      return page;
    },
  });`);
  source = replaceOnce(source, '      await campaignPresentationReady(page);', `      await campaignPresentationReady(page);
      const campaignRockId = await page.evaluate(() =>
        window.__rockRetirement.attach(window.__campaign.rendererOwner().world));`);
  source = replaceOnce(source,
    '      const launched = await page.evaluate(() => window.__campaign.fightReady());',
    `      const rockBeforeFight = await page.evaluate(() => {
        const id = window.__rockRetirement.attach(window.__campaign.rendererOwner().world);
        window.__rockRetirement.phase('battle');
        return id;
      });
      ctx.check(\`cycle \${cycle} campaign terrain updates reuse owned rock map\`,
        rockBeforeFight === campaignRockId, JSON.stringify({ campaignRockId, rockBeforeFight }));
      const launched = await page.evaluate(() => window.__campaign.fightReady());`);
  source = replaceOnce(source,
    '        const disposeRenderer = window.__game.disposeRenderer;',
    `        window.__rockRetirement.phase('campaign');
        const disposeRenderer = window.__game.disposeRenderer;`);
  source = replaceOnce(source,
    '      const retired = await page.evaluate(() => window.__rendererRetirement.sample());',
    `      const rock = await page.evaluate(() => window.__rockRetirement.sample());
      const retiredRock = rock.slice(0, cycle * 2);
      ctx.check(\`cycle \${cycle} owns one fresh rock bitmap per world\`,
        rock.length === cycle * 2 + 1 && rock.every((item, index) =>
          item.phase === (index % 2 === 0 ? 'campaign' : 'battle') && item.width > 0 && item.height > 0),
        JSON.stringify(rock));
      ctx.check(\`cycle \${cycle} disposes each retired rock texture and closes its bitmap exactly once\`,
        retiredRock.length === cycle * 2 && retiredRock.every(item => item.textureObserved &&
          item.disposals === 1 && item.closes === 1 && item.closedDimensions?.every(size => size === 0)) &&
          rock.at(-1)?.disposals === 0 && rock.at(-1)?.closes === 0,
        JSON.stringify(rock));
      ctx.check(\`cycle \${cycle} collects retired rock textures and decoded bitmaps\`,
        retiredRock.length === cycle * 2 && retiredRock.every(item => !item.textureAlive && !item.bitmapAlive),
        JSON.stringify(rock));
      const retired = await page.evaluate(() => window.__rendererRetirement.sample());`);
  source = replaceOnce(source, '        retired,', '        retired,\n        rock,');
  source = replaceOnce(source, '  } finally {\n    await page', `    if (process.env.ROCK_SETUP_FAILURE === '1') await (${setupFailure.toString()})(page, ctx);
  } finally {
    await page`);
  source = replaceOnce(source, '        window.__rendererRetirement?.restore();', `        window.__rockRetirement?.restore();
        delete window.__rockRetirement;
        window.__rendererRetirement?.restore();`);
  return source;
}

// Data URL preserves the original scene contract and absolute worlds import.
const derived = await import(`data:text/javascript;base64,${Buffer.from(await materialize()).toString('base64')}`);
export const meta = derived.meta;
export const run = derived.run;

if (process.argv[1] && pathToFileURL(process.argv[1]).href === import.meta.url) {
  if (process.argv[2] !== '--materialize' || !process.argv[3]) {
    throw new Error('Usage: node throwaway/rock-lifetime-proof.mjs --materialize OUTPUT.mjs');
  }
  await writeFile(process.argv[3], await materialize());
  console.log(fileURLToPath(pathToFileURL(process.argv[3])));
}

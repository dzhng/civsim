// Scratch measurement harness (throwaway). Decodes the real campaign raster in
// node and shims OffscreenCanvas so TerrainField can be built outside a browser.
import { readFileSync } from "node:fs";
import { PNG } from "../web/node_modules/pngjs/lib/png.js";

const ROOT = new URL("../", import.meta.url).pathname;

const png = PNG.sync.read(readFileSync(ROOT + "web/public/data/campaign-bg.png"));

class ShimCanvas {
  constructor(
    public width: number,
    public height: number,
  ) {}
  getContext() {
    const self = this;
    let pixels = new Uint8ClampedArray(self.width * self.height * 4);
    return {
      drawImage(src: { width: number; height: number; data: Uint8Array }, _x: number, _y: number) {
        // Box-average resample: the closest node stand-in for the browser's
        // smoothed drawImage. Exact when the sizes match.
        for (let y = 0; y < self.height; y++) {
          const sy0 = Math.floor((y * src.height) / self.height);
          const sy1 = Math.max(sy0 + 1, Math.floor(((y + 1) * src.height) / self.height));
          for (let x = 0; x < self.width; x++) {
            const sx0 = Math.floor((x * src.width) / self.width);
            const sx1 = Math.max(sx0 + 1, Math.floor(((x + 1) * src.width) / self.width));
            let r = 0,
              g = 0,
              b = 0,
              a = 0,
              n = 0;
            for (let sy = sy0; sy < sy1; sy++)
              for (let sx = sx0; sx < sx1; sx++) {
                const o = (sy * src.width + sx) * 4;
                r += src.data[o];
                g += src.data[o + 1];
                b += src.data[o + 2];
                a += src.data[o + 3];
                n++;
              }
            const d = (y * self.width + x) * 4;
            pixels[d] = r / n;
            pixels[d + 1] = g / n;
            pixels[d + 2] = b / n;
            pixels[d + 3] = a / n;
          }
        }
      },
      getImageData() {
        return { data: pixels };
      },
    };
  }
}
(globalThis as Record<string, unknown>).OffscreenCanvas = ShimCanvas;

const { TerrainField, TEMPERATE_Y_KM } = await import("../web/src/campaign/terrain.ts");

const map = JSON.parse(readFileSync(ROOT + "web/public/data/campaign-map.json", "utf8"));
const bgRect = JSON.parse(readFileSync(ROOT + "web/public/data/campaign-bg.json", "utf8"));
export const campaignData = {
  map,
  bg: { width: png.width, height: png.height, data: png.data },
  bgRect,
  nodeIndex: new Map(map.nodes.map((n: { id: number }, i: number) => [n.id, i])),
} as never;
export const field = new TerrainField(campaignData);
export { TEMPERATE_Y_KM };

/** A production-sized battle grid with irregular woods, rock, water and road. */
export function battleGrid() {
  const cell = 4,
    w = 300,
    h = 200;
  const tint = new Uint8Array(w * h);
  const height = new Float32Array(w * h);
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const i = y * w + x;
      const wood =
        Math.sin(x * 0.07) * Math.cos(y * 0.09) + Math.sin((x + y) * 0.031) * 0.7 > 0.55;
      tint[i] = wood ? 4 : x > 250 && y < 30 ? 1 : y === 120 ? 3 : 0;
      height[i] = Math.sin(x * 0.02) * 6 + Math.cos(y * 0.03) * 5;
    }
  return {
    w,
    h,
    cell,
    ox: (-w * cell) / 2,
    oy: (-h * cell) / 2,
    tint,
    height,
  };
}
